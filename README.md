\section{الاقتباس في البحث العلمي}

\begin{frame}{مفهوم الاقتباس في البحث العلمي}
    \begin{definition}
        \textbf{الاقتباس لغة:} أخذ الشيء بشكل كامل أو جزئي.
    \end{definition}
    
    \begin{block}{الاقتباس اصطلاحاً}
        هو أخذ جزء من مضمون مرجع ما وإلحاقه بمضمون البحث الحالي، مع التوثيق الكامل للمرجع الأساسي (اسم المؤلف، عنوان المرجع، دار النشر، سنة ومكان الصدور).
    \end{block}
\end{frame}

\begin{frame}{أهمية الاقتباس في البحث العلمي}
    \begin{enumerate}
        \item \textbf{الاحترافية:} يساعد الباحث في إعداد الدراسة بشكل احترافي.
        \item \textbf{توطيد الأفكار:} يساعد على فهم الأفكار وتأصيلها ونقدها موضوعياً.
        \item \textbf{استكشاف الآراء:} يساعد على التعرف على الآراء المعارضة والموالية للموضوع.
        \item \textbf{توضيح الرؤية:} يساعد على بيان وجهة نظر الباحث.
    \end{enumerate}
\end{frame}

\begin{frame}{أشكال الاقتباس (1)}
    \begin{description}
        \item[الاقتباس غير المباشر] نقل الفكرة بأسلوب الباحث مع الحفاظ على المعنى. لا تستخدم علامات التنصيص.
        \item[الاقتباس المباشر] نقل النص حرفياً كما هو:
        \begin{itemize}
            \item \textbf{القصير:} أقل من 40 كلمة، يوضع بين علامتي تنصيص
            \item \textbf{الطويل:} أكثر من 40 كلمة، يكتب كفقرة مستقلة
        \end{itemize}
    \end{description}
\end{frame}

\begin{frame}{أشكال الاقتباس (2)}
    \begin{description}
        \item[الاقتباس الجزئي] نقل جزء بسيط من جملة ودمجه في جملة الباحث.
        \item[الاقتباس المتقطع] حذف أجزاء غير ضرورية، ويُرمز لها بثلاث نقاط (...).
        \item[التلخيص] اختصار أفكار كبيرة في جمل بسيطة.
    \end{description}
\end{frame}

\begin{frame}{الضوابط والشروط}
    \begin{itemize}
        \item أن يفي الاقتباس بالمعنى الأصلي
        \item أن يكون مختصراً بقدر ما يستوفي المعنى
        \item الاقتباس من الأمور الضرورية والتركيز عليها
        \item أهمية التعقيب على الاقتباس
        \item الحفاظ على حقوق الملكية الفكرية
    \end{itemize}
\end{frame}

\begin{frame}{نسبة الاقتباس المسموحة}
    \begin{alertblock}{مهم}
        \begin{itemize}
            \item النسبة الشائعة: \textbf{20\% - 30\%} من إجمالي الدراسة
            \item تحسب من: الإطار النظري، الملخص، النتائج
            \item لا تدخل في الحساب: الخطة، قائمة المراجع، الفهارس
        \end{itemize}
    \end{alertblock}
\end{frame}

\begin{frame}{أخطاء شائعة}

        \end{itemize}
    \end{exampleblock}

\section{Pratique: Comment citer?}

\begin{frame}{Pourquoi citer des sources ?}
    \begin{itemize}
        \item Donner une \textbf{preuve scientifique}
        \item \textbf{Éviter le plagiat}
        \item Renforcer la \textbf{crédibilité}
        \item Aider à \textbf{retrouver} l'information
    \end{itemize}
\end{frame}

\begin{frame}{Comment choisir des références ?}
    \begin{enumerate}
        \item \textbf{Fiabilité scientifique}
        \item \textbf{Pertinence} avec le sujet
        \item \textbf{Qualité éditoriale}
    \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Comment citer en \LaTeX?}
    \begin{block}{Citation simple}
        \verb|\cite{mayers2003}|
    \end{block}
    \begin{block}{Citations multiples}
        \verb|\cite{demailly1991, mayers2003}|
    \end{block}
\end{frame}

\begin{frame}{Bonnes pratiques}
    \begin{itemize}
        \item Citer \textbf{dans le texte}, pas seulement en fin
        \item Garder un style \textbf{uniforme}
        \item Vérifier toutes les \textbf{informations}
    \end{itemize}
\end{frame}
